    2 pick           # Pick the condition
    run              # Run the condition on the right of the false branch
    not pick ldrop   # Drop the branch not chosen by the condition
    run              # Run the other branch
) map

'while' (
    (3 pick dup 4 throw run)
    (dup run while run)
    (ldrop ldrop)
    ifelse run
) map
